  ;; Set up the native Android GUI before loading the package-heavy config.
  ;; The GUI owns the Emacs server; NoD shells connect through its Unix socket.
  (when (eq system-type 'android)
    (setq temporary-file-directory
          (file-name-as-directory
           (expand-file-name "cache/nns" user-emacs-directory))
          create-lockfiles nil
          make-backup-files nil)
    (make-directory temporary-file-directory t)
    (add-to-list 'load-path (expand-file-name "lisp" user-emacs-directory))
    (require 'nns)
    (nns-mode 1)

    (require 'server)
    (setq server-socket-dir "/data/nix/var/emacs-server")
    ;; The directory belongs to Emacs, with search-only access for NoD.
    ;; Upstream rejects any group access, including search without write.
    (defun reomacs/android-server-ensure-safe-dir (original directory)
      "Accept only the provisioned, search-only Android server DIRECTORY."
      (if (equal (directory-file-name directory) server-socket-dir)
          (let ((attributes (file-attributes directory 'integer)))
            (unless (and (eq t (file-attribute-type attributes))
                         (= (file-attribute-user-id attributes) (user-uid))
                         (= (file-modes directory 'nofollow) #o2710))
              (error "Unsafe Android Emacs server directory: %s" directory)))
        (funcall original directory)))
    (when (file-directory-p server-socket-dir)
      (advice-add 'server-ensure-safe-dir :around
                  #'reomacs/android-server-ensure-safe-dir)
      (server-start)
      (set-file-modes (expand-file-name server-name server-socket-dir) #o770)))

  ;; Strategy: Always load pre-tangled config.el directly.
  ;; This avoids loading built-in org before straight can install a custom version.
  ;; Tangling happens automatically on save (see auto-tangle in config.org).
  ;; If config.el is missing, we fall back to tangling with built-in org.

  (let ((config-org (expand-file-name "config.org" user-emacs-directory))
        (config-el (expand-file-name "config.el" user-emacs-directory)))
    (cond
     ;; Prefer pre-tangled config.el (allows custom org version)
     ((file-exists-p config-el)
      (load-file config-el))
     ;; Fallback: tangle with built-in org (first-time bootstrap only)
     ((file-exists-p config-org)
      (require 'org)
      (let ((org-confirm-babel-evaluate nil))
        (org-babel-tangle-file config-org))
      (load-file config-el))
     (t
      (warn "Neither config.el nor config.org found in %s" user-emacs-directory))))

** Nix-managed binaries

Add binaries from nix flakes to Emacs' exec-path.
These are fetched and built by nix, then added to the PATH.

#+begin_src emacs-lisp
  (defun reomacs/nix-add-to-path (flake-ref package-path &optional callback)
    "Add a nix package binary directory to exec-path asynchronously.
  FLAKE-REF is the flake reference (e.g., 'github:numtide/nix-ai-tools').
  PACKAGE-PATH is the attribute path within the flake (e.g., 'codex-acp').
  CALLBACK is called with the store path on success, nil on failure."
    (let* ((installable (format "%s#%s^out" flake-ref package-path))
           (cmd (format "nix build %s --no-link --print-out-paths" installable))
           (buf (generate-new-buffer " *nix-add-to-path*")))
      (make-process
       :name "nix-add-to-path"
       :buffer buf
       :command (list "sh" "-c" cmd)
       :noquery t
       :sentinel
       (lambda (proc _event)
         (when (memq (process-status proc) '(exit signal))
           (unwind-protect
               (when (= (process-exit-status proc) 0)
                 (let* ((store-path (with-current-buffer (process-buffer proc)
                                      (string-trim (buffer-string))))
                        (bin-path (expand-file-name "bin" store-path)))
                   (when (and (not (string-empty-p store-path))
                              (file-directory-p bin-path))
                     (add-to-list 'exec-path bin-path)
                     (setenv "PATH" (concat bin-path ":" (getenv "PATH")))
                     (message "Added %s to PATH" store-path)
                     (when callback (funcall callback store-path)))))
             (when (buffer-live-p (process-buffer proc))
               (kill-buffer (process-buffer proc)))))))))
#+end_src

** ultra-scroll Instrumentation

#+begin_src emacs-lisp
  ;; ---------------------------------------------------------------------------
  ;; ultra-scroll Performance Instrumentation
  ;; ---------------------------------------------------------------------------

  (defvar reomacs/ultra-scroll--timing-ring (make-ring 100)
    "Ring buffer for scroll timing data (last 100 scrolls).")

  (defvar reomacs/ultra-scroll--timing-enabled nil
    "Whether scroll timing instrumentation is active.")

  (defun reomacs/ultra-scroll--wrap-with-timing (orig-fun delta)
    "Wrap scroll function ORIG-FUN with timing measurement.
DELTA is the pixel scroll amount."
    (if (not reomacs/ultra-scroll--timing-enabled)
        (funcall orig-fun delta)
      (let* ((start (current-time))
             (gc-before gc-elapsed)
             (gc-before-cons gcs-done)
             result)
        (setq result (funcall orig-fun delta))
        (let* ((elapsed (float-time (time-since start)))
               (gc-during (- gc-elapsed gc-before))
               (gc-cons-during (- gcs-done gc-before-cons)))
          (ring-insert reomacs/ultra-scroll--timing-ring
                       (list :function orig-fun
                             :delta delta
                             :elapsed elapsed
                             :gc-during gc-during
                             :gc-cons gc-cons-during
                             :buffer (buffer-name)
                             :major-mode major-mode
                             :time (current-time))))
        result)))

  (defun reomacs/ultra-scroll-timing-enable ()
    "Enable timing instrumentation for ultra-scroll functions."
    (interactive)
    (setq reomacs/ultra-scroll--timing-enabled t)
    (advice-add 'ultra-scroll-down :around #'reomacs/ultra-scroll--wrap-with-timing)
    (advice-add 'ultra-scroll-up :around #'reomacs/ultra-scroll--wrap-with-timing)
    (message "ultra-scroll timing instrumentation enabled"))

  (defun reomacs/ultra-scroll-timing-disable ()
    "Disable timing instrumentation for ultra-scroll functions."
    (interactive)
    (setq reomacs/ultra-scroll--timing-enabled nil)
    (advice-remove 'ultra-scroll-down #'reomacs/ultra-scroll--wrap-with-timing)
    (advice-remove 'ultra-scroll-up #'reomacs/ultra-scroll--wrap-with-timing)
    (message "ultra-scroll timing instrumentation disabled"))

  (defun reomacs/ultra-scroll-timing-report ()
    "Display timing report for recent scroll operations."
    (interactive)
    (if (ring-empty-p reomacs/ultra-scroll--timing-ring)
        (message "No timing data collected yet.")
      (let* ((items (ring-elements reomacs/ultra-scroll--timing-ring))
             (down-items (cl-remove-if-not (lambda (x) (eq (plist-get x :function) 'ultra-scroll-down)) items))
             (up-items (cl-remove-if-not (lambda (x) (eq (plist-get x :function) 'ultra-scroll-up)) items))
             (all-times (mapcar (lambda (x) (plist-get x :elapsed)) items))
             (gc-times (mapcar (lambda (x) (plist-get x :gc-during)) items))
             (avg-time (/ (cl-reduce #'+ all-times) (length all-times)))
             (max-time (apply #'max all-times))
             (min-time (apply #'min all-times))
             (avg-gc (/ (cl-reduce #'+ gc-times) (length gc-times))))
        (with-current-buffer (get-buffer-create "*ultra-scroll-timing*")
          (erase-buffer)
          (insert "== ultra-scroll Timing Report ==\n\n")
          (insert (format "Total scroll events: %d (down: %d, up: %d)\n"
                          (length items) (length down-items) (length up-items)))
          (insert (format "Avg time: %.3fms  Min: %.3fms  Max: %.3fms\n"
                          (* 1000 avg-time) (* 1000 min-time) (* 1000 max-time)))
          (insert (format "Avg GC time during scroll: %.3fms\n\n" (* 1000 avg-gc)))
          (insert "Distribution:\n")
          (let ((ranges '((0 0.003 "<3ms") (0.003 0.005 "3-5ms") (0.005 0.010 "5-10ms")
                          (0.010 0.015 "10-15ms") (0.015 0.020 "15-20ms") (0.020 999 ">20ms"))))
            (dolist (range ranges)
              (let* ((min (nth 0 range))
                     (max (nth 1 range))
                     (label (nth 2 range))
                     (count (cl-count-if (lambda (tm) (and (>= tm min) (< tm max))) all-times)))
                (insert (format "  %5s: %s\n" label (make-string count ?█))))))
          (insert "\nRecent slow scrolls (>10ms):\n")
          (let ((slow (cl-remove-if-not (lambda (x) (> (plist-get x :elapsed) 0.010)) items))
                (cnt 0))
            (dolist (s (reverse slow))
              (when (< cnt 10)

  (defun reomacs/scroll-jitter-report ()
    "Analyze scroll timing for jitter (irregular intervals)."
    (interactive)
    (if (< (length reomacs/scroll--jitter-log) 10)
        (message "Not enough data. Scroll more first.")
      (let* ((times (reverse reomacs/scroll--jitter-log))
             (intervals nil)
             prev-time)
        (dolist (entry times)
          (let ((ts (car entry)))
            (when prev-time
              (push (float-time (time-subtract ts prev-time)) intervals))
            (setq prev-time ts)))
        (let* ((avg-int (/ (cl-reduce #'+ intervals) (length intervals)))
               (min-int (apply #'min intervals))
               (max-int (apply #'max intervals))
               (jitter (- max-int min-int))
               (cv (/ (sqrt (/ (cl-reduce #'+ (mapcar (lambda (x) (expt (- x avg-int) 2)) intervals))
                               (length intervals)))
                      avg-int)))
          (with-current-buffer (get-buffer-create "*scroll-jitter*")
            (erase-buffer)
            (insert "== Scroll Smoothness Analysis ==\n\n")
            (insert (format "Display: Assumed 160Hz (target: %.2fms per frame)\n\n" (/ 1000.0 160)))
            (insert (format "Avg interval: %.3fms\n" (* 1000 avg-int)))
            (insert (format "Min interval: %.3fms\n" (* 1000 min-int)))
            (insert (format "Max interval: %.3fms\n" (* 1000 max-int)))
            (insert (format "Jitter (max-min): %.3fms\n\n" (* 1000 jitter)))
            (insert (format "Coefficient of variation: %.1f%%\n\n" (* 100 cv)))
            (insert "Interpretation:\n")
            (cond
             ((> cv 0.5) (insert "  🔴 High jitter (>50%) - very noticeable stutter\n"))
             ((> cv 0.3) (insert "  🟡 Moderate jitter (30-50%) - some stutter visible\n"))
             ((> cv 0.15) (insert "  🟡 Low jitter (15-30%) - minor stutter\n"))
             (t (insert "  ✅ Very smooth (<15%) - excellent\n")))
            (insert "\n" (if (> (* 1000 avg-int) 10)
                        "⚠️  Intervals >10ms suggest event batching or frame drops"
                      "✓ Intervals look good for high-refresh"))
            (pop-to-buffer (current-buffer)))))))

  (defun reomacs/scroll--detect-pause (orig-fun delta)
    "Detect and log what's causing long pauses during scroll."
    (let* ((now (current-time))
           (pause-duration (when reomacs/scroll--last-event-time
                             (float-time (time-subtract now reomacs/scroll--last-event-time)))))
      ;; Log if this is a significant pause (>50ms)
      (when (and pause-duration (> pause-duration reomacs/scroll--pause-threshold))
        (let* ((gc-before gc-elapsed)
               (gc-before-cons gcs-done)
               (gc-during 0)
               culprit)
          ;; Try to identify the culprit
          (setq culprit
                (cond
                 ((and (> (- gc-elapsed gc-before) 0)
                       (> (- gc-elapsed gc-before) (* pause-duration 0.5)))
                  (format "GC (%.0f%% of pause)" (* 100 (/ (- gc-elapsed gc-before) pause-duration))))
                 ((and (> (- gc-elapsed gc-before) 0) (> pause-duration 0.1))
                  (format "GC+other (%.0f%%)" (* 100 (/ (- gc-elapsed gc-before) pause-duration))))
                 ((bound-and-true-p jit-lock--defer-timer) "jit-lock-defer")
                 ((bound-and-true-p company-pseudo-tooltip-overlay) "company-tooltip")
                 ((and (boundp 'lsp--buffer-workspaces) lsp--buffer-workspaces) "LSP")
                 ((and (boundp 'flycheck-mode) flycheck-mode) "flycheck")
                 ((and (boundp 'flymake-mode) flymake-mode) "flymake")
                 ((and (boundp 'font-lock-mode) font-lock-mode) "font-lock")
                 ((> (length post-command-hook) 10) "heavy-post-command")
                 ((> (length pre-redisplay-functions) 5) "heavy-redisplay")
                 ((and (boundp 'undo-tree-mode) undo-tree-mode) "undo-tree")
                 (t "unknown")))
          (push (list now pause-duration culprit delta (buffer-name) major-mode)
                reomacs/scroll--pause-log)
          ;; Keep last 50 pauses
          (when (> (length reomacs/scroll--pause-log) 50)
            (setq reomacs/scroll--pause-log (cl-subseq reomacs/scroll--pause-log 0 50)))))
      (setq reomacs/scroll--last-event-time now)
      (funcall orig-fun delta)))

      "Like `vertico-directory-delete-char' but respects TRAMP boundaries.
When in a remote file path, treats `:' `@' `|' and `#' as additional
boundaries beyond `/', so backspace stops at TRAMP components.
If point is right after a delimiter, delete just that delimiter.
If point is after a non-delimiter, delete back to the previous boundary."
      (interactive "p")
      (let* ((file-category (eq 'file (vertico--metadata-get 'category)))
             (no-region (not (and (use-region-p) delete-active-region (= n 1))))
             (in-remote (and file-category
                             no-region
                             (> (point) (minibuffer-prompt-end))
                             (vertico--remote-p
                              (buffer-substring-no-properties
                               (minibuffer-prompt-end) (point))))))
        (cond
         ;; TRAMP path, at a delimiter: delete just the delimiter
         ((and in-remote (memq (char-before) '(?/ ?: ?@ ?| ?#)))
          (with-no-warnings (delete-backward-char n)))
         ;; TRAMP path, at a non-delimiter: delete back to previous boundary
         ((and in-remote (> (point) (minibuffer-prompt-end)))
          (let (found)
            (dotimes (_ n found)
              (save-excursion
                (let ((end (point)))
                  (goto-char (1- end))
                  (if (re-search-backward "[/:@|#]" (minibuffer-prompt-end) t)
                      (progn
                        (delete-region (1+ (point)) end)
                        (setq found t))
                    (delete-region (minibuffer-prompt-end) end)
                    (setq found t)))))
            (unless found
              (with-no-warnings (delete-backward-char n)))))
         ;; Non-remote: default vertico-directory behavior
         (t
          (unless (and no-region (eq (char-before) ?/)
                       (vertico-directory-up n))
            (with-no-warnings (delete-backward-char n))))))))

    "Return the current project root, or signal outside a project."
    (or (when-let* ((project (project-current)))
          (project-root project))
        (user-error "Not inside a project")))

  (defun reomacs/azdo-pr--jj (&rest args)
    "Run jj with ARGS in the current project and return cleaned output."
    (let ((default-directory (reomacs/azdo-pr--root)))
      (with-temp-buffer
        (when (zerop (apply #'call-process "jj" nil t nil args))
          (require 'ansi-color)
          (ansi-color-filter-apply (string-trim (buffer-string)))))))

    "Infer Azure DevOps repository settings from the jj `origin' remote."
    (let* ((remote-lines (split-string (or (reomacs/azdo-pr--jj "git" "remote" "list") "") "\n" t))
           (remote (catch 'remote
                     (dolist (line remote-lines)
                       (when (string-match "\\`origin[[:space:]]+\\(.+\\)\\'" line)
                         (throw 'remote (match-string 1 line)))))))
      (unless remote
        (user-error "No jj origin remote found"))
      (cond
       ((string-match "\\`https?://\\(?:[^/@]+@\\)?dev\\.azure\\.com/\\([^/]+\\)/\\([^/]+\\)/_git/\\([^/?#]+\\)\\(?:\\.git\\)?/?\\'" remote)
        (list :organization (format "https://dev.azure.com/%s" (match-string 1 remote))
              :project (match-string 2 remote)
              :repository (match-string 3 remote)))
       ((string-match "\\`https?://\\([^./]+\\)\\.visualstudio\\.com/\\([^/]+\\)/_git/\\([^/?#]+\\)\\(?:\\.git\\)?/?\\'" remote)
        (list :organization (format "https://%s.visualstudio.com" (match-string 1 remote))
              :project (match-string 2 remote)
              :repository (match-string 3 remote)))
       ((string-match "\\`git@ssh\\.dev\\.azure\\.com:v3/\\([^/]+\\)/\\([^/]+\\)/\\([^/?#]+\\)\\(?:\\.git\\)?/?\\'" remote)
        (list :organization (format "https://dev.azure.com/%s" (match-string 1 remote))
              :project (match-string 2 remote)
              :repository (match-string 3 remote)))

** Column visibility

Toggle individual columns in dired via overlays with =invisible= property.
All columns are always present in the buffer (=-alh= switches); visibility
is controlled per-column through =buffer-invisibility-spec=.

#+begin_src emacs-lisp

  (defvar reomacs/dired-column-specs
    '((perms . reomacs/dired-hide-perms)
      (links . reomacs/dired-hide-links)
      (owner . reomacs/dired-hide-owner)
      (group . reomacs/dired-hide-group)
      (size  . reomacs/dired-hide-size)
      (date  . reomacs/dired-hide-date))

    (concat
     "."                                           ; mark
     "\\( [-drwxlsStT]\\{10\\}[.+@]?\\)"           ; 1: perms (no space in ACL class)
     "\\( *[0-9]+\\)"                              ; 2: links (0+ spaces for high counts)
     "\\( +[^ ]+\\)"                               ; 3: owner
     "\\( +[^ ]+\\)"                               ; 4: group
     "\\( +[^ ]+\\)"                               ; 5: size
     )

    "Add invisible overlays for each column on file lines between BEG and END."
    (save-excursion
      (goto-char beg)
      (while (and (not (eobp)) (< (point) end))
        (let ((fn-pos (save-excursion (dired-move-to-filename nil))))
          (when fn-pos
            (beginning-of-line)
            (when (looking-at reomacs/dired-column-re)
              (let ((col-names '(perms links owner group size)))
                (cl-loop for name in col-names
                         for i from 1
                         do (let ((ov (make-overlay (match-beginning i)
                                                    (match-end i))))
                              (overlay-put ov 'invisible
                                           (alist-get name reomacs/dired-column-specs))
                              (overlay-put ov 'reomacs/dired-column t)
                              (overlay-put ov 'evaporate t))))
              (let ((ov (make-overlay (match-end 5) fn-pos)))
                (overlay-put ov 'invisible
                             (alist-get 'date reomacs/dired-column-specs))
                (overlay-put ov 'reomacs/dired-column t)
                (overlay-put ov 'evaporate t)))))
        (forward-line 1))))

(defun reomacs/wdired-create-new-files-advice (orig-fun &rest args)
  "Advise `wdired-finish-edit` to create new files and refresh.

This function is designed to be used as `:around` advice. It scans the
`wdired` buffer for lines that do not correspond to existing file
entries (i.e., they lack the `dired-filename` text property).

For each such line found:
1. It parses the line as a new file path.
2. It creates any non-existent parent directories recursively.
3. It creates the new file (as an empty file).
4. It removes the line from the buffer.

After creating all new files, it calls the original `wdired-finish-edit`
function. If any files were created by this advice, it then calls
`revert-buffer` to refresh the dired listing."
  (let ((files-created-p nil))
    (save-excursion
      ;; Go to the beginning of the buffer to start our scan.
      (goto-char (point-min))

      ;; 1. First, we collect all the lines that represent new files.
      ;; We do this first to avoid issues with modifying the buffer while iterating over it.
      (let ((new-entries '()))
        (while (not (eobp))
          (let ((line-start (line-beginning-position))
                (line-end (line-end-position)))
            ;; A line for a new file won't have the 'dired-filename' property.
            (unless (get-text-property line-start 'dired-filename)
              (let ((filename (string-trim (buffer-substring-no-properties line-start line-end))))
                ;; We only care about lines that actually contain non-whitespace text.
                (when (string-match-p "\\S-" filename)
                  ;; We push the filename and its start/end positions onto the list.
                  (push (list filename line-start line-end) new-entries))))
            (forward-line 1)))

        ;; 2. Now, process the collected new files.
        ;; We iterate through the list we just built. The list is naturally in
        ;; reverse order of appearance, which is perfect for deleting lines
        ;; from the buffer without invalidating the position markers of subsequent items.
        (when new-entries
          (setq files-created-p t)) ; Mark that we are creating files.
        (dolist (entry new-entries)
          (let* ((filename (nth 0 entry))
                 (start-pos (nth 1 entry))
                 (end-pos (nth 2 entry))
                 ;; Construct the full path relative to the dired buffer's directory.
                 (full-path (expand-file-name filename default-directory))
                 (directory (file-name-directory full-path)))

            ;; Create parent directories if they don't already exist.
            ;; The 't' argument to `make-directory` means "create parents as needed".
            (when (and directory (not (file-directory-p directory)))
              (make-directory directory t)
              (message "Created directory: %s" (file-name-sans-trailing-slash directory)))

            ;; Create the file itself if it doesn't exist.
            ;; Using `with-temp-buffer` and `write-file` is a clean way to "touch" a file.
            (unless (file-exists-p full-path)
              (with-temp-buffer
                (write-file full-path))
              (message "Created file: %s" full-path))

            ;; Finally, remove the processed line from the wdired buffer so the original
            ;; function doesn't see it and get confused.
            (goto-char start-pos)
            ;; We delete the region including the newline character.
            (delete-region start-pos (1+ end-pos)))))))

    ;; 3. With our work done, call the original function to let it handle its business.
    (apply orig-fun args)

    ;; 4. If we created any files, refresh the dired buffer to show them.
    (when files-created-p
      (revert-buffer)
      (message "Dired buffer refreshed to show new files.")))

;; Add our function as "around" advice to `wdired-finish-edit`.
;; This means our function wraps the original one.
(advice-add 'wdired-finish-edit :around #'reomacs/wdired-create-new-files-advice)

  (defun reomodeline--jj-start-process ()
    "Start async jj process to get current change id."
    (when-let* ((root (reomodeline--jj-root)))
      ;; Cancel any existing process
      (when (process-live-p reomodeline--jj-proc)
        (delete-process reomodeline--jj-proc))
      (let* ((default-directory root)
             (buf (current-buffer))
             (proc-buf (generate-new-buffer " *reomodeline-jj*")))
        (setq reomodeline--jj-proc
              (make-process
               :name "reomodeline-jj" :buffer proc-buf :noquery t
               :command '("jj" "log" "-r" "@" "--no-graph" "-T" "change_id.short(8)")
               :sentinel
               (lambda (proc _event)
                 (when (memq (process-status proc) '(exit signal))
                   (unwind-protect
                       (when (and (= (process-exit-status proc) 0)
                                  (buffer-live-p buf))
                         (let ((output (with-current-buffer (process-buffer proc)
                                         (string-trim (buffer-string)))))
                           (with-current-buffer buf
                             (setq reomodeline--jj-info
                                   (unless (string-empty-p output) output))
                             (force-mode-line-update))))
                     (when (buffer-live-p (process-buffer proc))
                       (kill-buffer (process-buffer proc)))))))))))

** Basic Settings

#+begin_src emacs-lisp
  ;; Prevent TRAMP from creating extra files and use scp directly when moving files
  (setq remote-file-name-inhibit-locks t
        tramp-use-scp-direct-remote-copying t
        remote-file-name-inhibit-auto-save-visited t
        tramp-verbose 1
        tramp-remote-shell "/bin/sh"
        tramp-use-ssh-controlmaster-options nil
        ;; Prevent infinite recursion on FTP paths
        max-lisp-eval-depth 3000)

  ;; Override SetEnv TERM=xterm-256color from ~/.ssh/config.
  ;; TRAMP needs TERM=dumb so the remote shell uses a plain prompt
  ;; that TRAMP can pattern-match; xterm-256color causes fancy/colored
  ;; prompts that hang the connection handshake.
  (with-eval-after-load 'tramp
    (let ((login-args (assoc 'tramp-login-args
                             (cdr (assoc "ssh" tramp-methods)))))
      (when login-args
        (setcdr login-args
                (append (cdr login-args)
                        '(("-o" "SetEnv=TERM=dumb")))))))

  (when (eq system-type 'android)
    (with-eval-after-load 'tramp
      (let* ((method (copy-tree (cdr (assoc "ssh" tramp-methods))))
             (login-args (assoc 'tramp-login-args method))
             (ssh-directory (expand-file-name ".ssh" "~")))
        ;; The NNS launcher needs its preload, but Android's SSH linker forbids it.
        (setcar (cdr (assoc 'tramp-login-program method)) "/system/bin/env")
        ;; Android's /bin/sh replaces the prompt TRAMP uses to detect readiness.
        (setcar (cdr (assoc 'tramp-remote-shell method))
                "/nix/var/nix/profiles/default/bin/bash")
        (setcar (cdr login-args)
                (append '(("-u" "LD_PRELOAD") ("/product/bin/ssh"))
                        (butlast (cadr login-args))
                        `(("-o" "BatchMode=yes")
                          ("-o" "ConnectTimeout=5")
                          ("-o" "GlobalKnownHostsFile=/dev/null")
                          ("-o" ,(concat "UserKnownHostsFile="
                                          (expand-file-name "known_hosts" ssh-directory)))
                          ("-i" ,(expand-file-name "id_ed25519" ssh-directory))
                          ("-o" "StrictHostKeyChecking=yes"))
                        (last (cadr login-args))))
        (push (cons "nod" method) tramp-methods))))

  ;; Fail fast on unreachable hosts instead of hanging for 30-60s.
  ;; TRAMP-level timeout (seconds to wait for connection).
  (setq tramp-connection-timeout 5)

  ;; Don't search auth-sources during TRAMP completion.
  ;; Default behaviour decrypts ~/.authinfo.gpg to list known hosts,
  ;; which silently blocks waiting for YubiKey touch inside vertico.
  (setq tramp-completion-use-auth-sources nil)
#+end_src

** Caching for Remote Operations

Memoize expensive TRAMP operations to avoid repeated remote calls.

#+begin_src emacs-lisp
  (defun memoize-remote (key cache orig-fn &rest args)
    "Memoize a value if the key is a remote path."
    (if (and key (file-remote-p key))
        (if-let* ((current (assoc key (symbol-value cache))))
            (cdr current)
          (let ((current (apply orig-fn args)))
            (set cache (cons (cons key current) (symbol-value cache)))
            current))
      (apply orig-fn args)))

  ;; Memoize project-current
  (defvar project-current-cache nil)
  (defun memoize-project-current (orig &optional prompt directory)
    (let ((dir (or directory project-current-directory-override default-directory)))
      ;; Skip for non-git remote protocols (ftp, dav, etc)
      (if (and (file-remote-p dir)
               (not (string-match "^/\\(ssh\\|scp\\|sftp\\):" dir)))
          nil  ; Return nil for non-git remotes to avoid git lookups
        (memoize-remote dir 'project-current-cache orig prompt directory))))
  (advice-add 'project-current :around #'memoize-project-current)

  ;; Memoize magit-toplevel
  (defvar magit-toplevel-cache nil)
  (defun memoize-magit-toplevel (orig &optional directory)
    (let ((dir (or directory default-directory)))
      ;; Skip for non-git remote protocols
      (if (and (file-remote-p dir)
               (not (string-match "^/\\(ssh\\|scp\\|sftp\\):" dir)))
          nil
        (memoize-remote dir 'magit-toplevel-cache orig directory))))
  (advice-add 'magit-toplevel :around #'memoize-magit-toplevel)

  ;; Memoize vc-git-root
  (defvar vc-git-root-cache nil)
  (defun memoize-vc-git-root (orig file)
    (let ((dir (file-name-directory file)))
      ;; Skip for non-git remote protocols
      (if (and (file-remote-p dir)
               (not (string-match "^/\\(ssh\\|scp\\|sftp\\):" dir)))
          nil
        (let ((value (memoize-remote dir 'vc-git-root-cache orig file)))
          ;; Sometimes vc-git-root returns nil even when there is a root there
          (when (null (cdr (car vc-git-root-cache)))
            (setq vc-git-root-cache (cdr vc-git-root-cache)))
          value))))
  (advice-add 'vc-git-root :around #'memoize-vc-git-root)

  ;; Reset caches when needed
  (defun reomacs/reset-tramp-caches ()
    "Reset all TRAMP memoization caches."
    (interactive)
    (setq project-current-cache nil
          magit-toplevel-cache nil
          vc-git-root-cache nil)
    (message "TRAMP caches reset"))
#+end_src
