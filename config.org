  ;; Check minimum Emacs version (29+ for treesit, 30+ for some features)
  (when (< emacs-major-version 29)
    (error "Reomacs requires Emacs 29+, you have %s" emacs-version))

  ;; Strategy: Always load pre-tangled config.el directly.
  ;; This avoids loading built-in org before straight can install a custom version.
  ;; Tangling happens automatically on save (see auto-tangle in config.org).
  ;; If config.el is missing, we fall back to tangling with built-in org.

  (let ((config-org (expand-file-name "config.org" user-emacs-directory))
        (config-el (expand-file-name "config.el" user-emacs-directory)))
    (cond
     ;; Prefer pre-tangled config.el (allows custom org version)
     ((file-exists-p config-el)
      (load-file config-el))
     ;; Fallback: tangle with built-in org (first-time bootstrap only)
     ((file-exists-p config-org)
      (require 'org)
      (let ((org-confirm-babel-evaluate nil))
        (org-babel-tangle-file config-org))
      (load-file config-el))
     (t
      (warn "Neither config.el nor config.org found in %s" user-emacs-directory))))

    "Tangle config.org automatically if in `user-emacs-directory`"
    (when (string-equal (buffer-file-name)
                        (expand-file-name "config.org" user-emacs-directory))
      (let ((org-confirm-babel-evaluate nil))
        (org-babel-tangle))))

    "Functions to run after init and after config reload.")

    "Run all functions in `reomacs/post-init-hooks'."
    (run-hooks 'reomacs/post-init-hooks))

  (add-hook 'emacs-startup-hook #'reomacs/run-post-init-hooks)

  (defvar reomacs/config-loaded-p nil
    "Non-nil after initial config load. Used to skip re-initialization.")

  (defun reomacs/reload-config ()
    "Tangle config.org and reload config.el.
Skips display-touching initialization on reload to avoid PGTK crashes."
    (interactive)
    (let ((org-confirm-babel-evaluate nil)
          (reomacs/config-loaded-p t))  ; Signal we're reloading
      (org-babel-tangle-file (expand-file-name "config.org" user-emacs-directory))
      (load-file (expand-file-name "config.el" user-emacs-directory))
      (reomacs/run-post-init-hooks)
      (message "Config tangled and reloaded.")))

  (add-hook 'emacs-startup-hook (lambda () (setq reomacs/config-loaded-p t)) 100)

** Nix-managed binaries

Add binaries from nix flakes to Emacs' exec-path.
These are fetched and built by nix, then added to the PATH.

#+begin_src emacs-lisp
  (defun reomacs/nix-add-to-path (flake-ref package-path &optional callback)
    "Add a nix package binary directory to exec-path asynchronously.
  FLAKE-REF is the flake reference (e.g., 'github:numtide/nix-ai-tools').
  PACKAGE-PATH is the attribute path within the flake (e.g., 'codex-acp').
  CALLBACK is called with the store path on success, nil on failure."
    (let* ((installable (format "%s#%s" flake-ref package-path))
           (cmd (format "nix build %s --no-link --print-out-paths" installable))
           (buf (generate-new-buffer " *nix-add-to-path*")))
      (make-process
       :name "nix-add-to-path"
       :buffer buf
       :command (list "sh" "-c" cmd)
       :noquery t
       :sentinel
       (lambda (proc _event)
         (when (memq (process-status proc) '(exit signal))
           (unwind-protect
               (when (= (process-exit-status proc) 0)
                 (let* ((store-path (with-current-buffer (process-buffer proc)
                                      (string-trim (buffer-string))))
                        (bin-path (expand-file-name "bin" store-path)))
                   (when (and (not (string-empty-p store-path))
                              (file-directory-p bin-path))
                     (add-to-list 'exec-path bin-path)
                     (setenv "PATH" (concat bin-path ":" (getenv "PATH")))
                     (message "Added %s to PATH" store-path)
                     (when callback (funcall callback store-path)))))
             (when (buffer-live-p (process-buffer proc))
               (kill-buffer (process-buffer proc)))))))))
#+end_src

** Treesit Rainbow Delimiters

Tree-sitter based rainbow delimiters - works for all languages with tree-sitter grammars.
Uses incremental updates for performance. Universal across Lisp, C-like, Script, and Data languages.

#+begin_src emacs-lisp
  ;;; Treesit Rainbow Delimiters - Universal rainbow highlighting using tree-sitter
  ;; Based on nvim's rainbow-delimiters.nvim

  (defvar reomacs/treesit-rainbow--overlays nil
    "List of overlays for current buffer.")

  (defvar reomacs/treesit-rainbow--parser nil
    "Tree-sitter parser for current buffer.")

  (defvar-local reomacs/treesit-rainbow--org-parsers nil
    "Treesit parsers created for org-mode source blocks.")

  (defvar reomacs/treesit-rainbow--depth 0
    "Current nesting depth during tree traversal.")

  (defvar reomacs/treesit-rainbow--colors
    (list "#f7768e" "#e0af68" "#7aa2f7" "#ff9e64" "#9d7cd8" "#7dcfff")
    "Rainbow color palette.")

  (defvar reomacs/treesit-rainbow--num-colors 6
    "Number of colors in the palette.")

  ;; Query alist: language -> list of node types to match
  ;; Each entry is (container-type (open-delims) (close-delims) [child-predicate])
  ;; child-predicate: optional function symbol, children matching it are colored at depth+1
  (defvar reomacs/treesit-rainbow--queries
    '((elisp . ((list . ((["("]) ([")"])  reomacs/treesit-rainbow--elisp-keyword-p))
                (special_form . ((["("]) ([")"])  reomacs/treesit-rainbow--elisp-keyword-p))
                (function_definition . ((["("]) ([")"])  reomacs/treesit-rainbow--elisp-keyword-p))
                (macro_definition . ((["("]) ([")"])  reomacs/treesit-rainbow--elisp-keyword-p))
                (vector . ((["["]) (["]"])))
                (string . ((["\""]) (["\""])))))
      (fennel . ((list . ((["("]) ([")"])))
                 (table . ((["{"]) (["}"])))))
      (clojure . ((list_lit . ((["("]) ([")"])))
                  (vec_lit . ((["["]) (["]"])))
                  (map_lit . ((["{"]) (["}"])))
                  (set_lit . ((["{"]) (["}"])))))
      (scheme . ((list . ((["("]) ([")"])))))
      (c . ((compound_statement . ((["{"]) (["}"])))
            (argument_list . ((["("]) ([")"])))
            (parameter_list . ((["("]) ([")"])))
            (parenthesized_expression . ((["("]) ([")"])))
            (subscript_expression . ((["["]) (["]"])))

      (cpp . ((compound_statement . ((["{"]) (["}"])))
              (argument_list . ((["("]) ([")"])))
              (parameter_list . ((["("]) ([")"])))
              (parenthesized_expression . ((["("]) ([")"])))
              (subscript_expression . ((["["]) (["]"])))

               (tuple_pattern . ((["("]) ([")"])))))
      (go . ((block . ((["{"]) (["}"])))
             (argument_list . ((["("]) ([")"])))
             (parameter_list . ((["("]) ([")"])))
             (literal_value . ((["{"]) (["}"])))
             (composite_literal . ((["{"]) (["}"])))
             (array_type . ((["["]) (["]"])))
             (slice_type . ((["["]) (["]"])))

      (toml . ((inline_table . ((["{"]) (["}"])))
               (array . ((["["]) (["]"]))))))
    "Alist of language -> ((container-type . (open-delims) (close-delims))) for rainbow delimiters.")

  ;; Get language for current buffer
  (defun reomacs/treesit-rainbow--get-language ()
    "Get tree-sitter language for current buffer."
    (or (treesit-language-at (point-min))
        (when (bound-and-true-p treesit--language)
          treesit--language)))

  (defun reomacs/treesit-rainbow--get-query (lang)
    "Get delimiter query for language LANG."
    (cdr (or (assoc lang reomacs/treesit-rainbow--queries)
             (assoc 'elisp reomacs/treesit-rainbow--queries))))

  (defun reomacs/treesit-rainbow--get-face (depth)
    "Get face for DEPTH using rainbow color palette."
    (let* ((idx (mod (1- depth) reomacs/treesit-rainbow--num-colors))
           (color (nth idx reomacs/treesit-rainbow--colors))
           (face-name (intern (format "reomacs/treesit-rainbow-depth-%d" (1+ idx)))))
      (unless (facep face-name)
        (make-face face-name)
        (set-face-foreground face-name color))
      face-name))

  (defun reomacs/treesit-rainbow--apply-delimiter (node)
    "Apply rainbow overlay to a delimiter NODE."
    (let* ((beg (treesit-node-start node))
           (end (treesit-node-end node))
           (depth reomacs/treesit-rainbow--depth)
           (face (reomacs/treesit-rainbow--get-face depth)))
      (when (> end beg)
        (let ((overlay (make-overlay beg end)))
          (overlay-put overlay 'face face)
          (overlay-put overlay 'priority 100)
          (push overlay reomacs/treesit-rainbow--overlays)))))

  (defun reomacs/treesit-rainbow--elisp-keyword-p (node)
    "Return non-nil if NODE is an elisp keyword (symbol starting with :)."
    (and (string= (treesit-node-type node) "symbol")
         (string-prefix-p ":" (treesit-node-text node))))

  (defun reomacs/treesit-rainbow--delim-string (v)
    "Extract delimiter string from V (may be a vector or string)."
    (if (vectorp v) (aref v 0) v))

  (defun reomacs/treesit-rainbow--find-delimiters (container-node container-type)
    "Find delimiter nodes within CONTAINER-NODE of CONTAINER-TYPE.
QUERY-SPEC format: (open-delims close-delims [child-predicate]).
When CHILD-PREDICATE is a function symbol, matching children are
colored one level deeper than the container delimiters."
    (let* ((query-spec (cdr (assoc (intern container-type)
                                   (reomacs/treesit-rainbow--get-query
                                    (treesit-parser-language reomacs/treesit-rainbow--parser))))))
      (when query-spec
        (let* ((open-delims (mapcar #'reomacs/treesit-rainbow--delim-string (car query-spec)))
               (close-delims (mapcar #'reomacs/treesit-rainbow--delim-string (cadr query-spec)))
               (child-pred (caddr query-spec)))
          ;; Search for matching delimiter nodes
          (let ((child (treesit-node-child container-node 0)))
            (while child
              (let ((text (treesit-node-text child)))
                (cond
                 ;; Match opening delimiter
                 ((member text open-delims)
                  (reomacs/treesit-rainbow--apply-delimiter child))
                 ;; Match closing delimiter
                 ((member text close-delims)
                  (reomacs/treesit-rainbow--apply-delimiter child))
                 ;; Match child predicate at depth+1
                 ((and child-pred (funcall child-pred child))
                  (let ((reomacs/treesit-rainbow--depth (1+ reomacs/treesit-rainbow--depth)))
                    (reomacs/treesit-rainbow--apply-delimiter child)))))
              (setq child (treesit-node-next-sibling child))))))))

  (defun reomacs/treesit-rainbow--traverse (node)
    "Recursively traverse parse tree NODE and apply overlays."
    (let* ((type (treesit-node-type node))
           (type-sym (intern type)))
      ;; Check if this node type is a container we should highlight
      (when (assoc type-sym (reomacs/treesit-rainbow--get-query
                             (treesit-parser-language reomacs/treesit-rainbow--parser)))
        (cl-incf reomacs/treesit-rainbow--depth)
        (reomacs/treesit-rainbow--find-delimiters node type)
        ;; Traverse children
        (let ((child (treesit-node-child node 0)))
          (while child
            (reomacs/treesit-rainbow--traverse child)
            (setq child (treesit-node-next-sibling child))))
        (cl-decf reomacs/treesit-rainbow--depth))
      ;; Also traverse even if not a container (for nested structures)
      (unless (assoc type-sym (reomacs/treesit-rainbow--get-query
                               (treesit-parser-language reomacs/treesit-rainbow--parser)))
        (let ((child (treesit-node-child node 0)))
          (while child
            (reomacs/treesit-rainbow--traverse child)
            (setq child (treesit-node-next-sibling child)))))))

  (defun reomacs/treesit-rainbow--org-lang-to-treesit (lang-str)
    "Map org source block language name LANG-STR to treesit language symbol."
    (intern (or (cdr (assoc lang-str '(("emacs-lisp" . "elisp")
                                        ("elisp" . "elisp")
                                        ("lisp" . "elisp"))))
                lang-str)))

  (defun reomacs/treesit-rainbow--highlight-org-blocks ()
    "Apply rainbow overlays to source blocks in org-mode."
    (dolist (parser reomacs/treesit-rainbow--org-parsers)
      (treesit-parser-delete parser))
    (setq reomacs/treesit-rainbow--org-parsers nil)
    (let ((blocks-by-lang (make-hash-table :test 'eq)))
      ;; Collect source block ranges grouped by language
      (org-element-map (org-element-parse-buffer) 'src-block
        (lambda (block)
          (let* ((lang-str (org-element-property :language block))
                 (lang (and lang-str
                            (reomacs/treesit-rainbow--org-lang-to-treesit lang-str))))
            (when (and lang
                       (assoc lang reomacs/treesit-rainbow--queries)
                       (treesit-language-available-p lang))
              (let* ((content-beg (save-excursion
                                    (goto-char (org-element-property :post-affiliated block))
                                    (forward-line 1)
                                    (point)))
                     (content-end (save-excursion
                                    (goto-char (org-element-property :end block))
                                    (when (re-search-backward
                                           "^[ \t]*#\\+end_src"
                                           (org-element-property :begin block) t)
                                      (point)))))
                (when (and content-end (> content-end content-beg))
                  (push (cons content-beg content-end)
                        (gethash lang blocks-by-lang))))))))
      ;; Create a parser per language with all its block ranges
      (maphash
       (lambda (lang ranges)
         (let ((parser (treesit-parser-create lang)))
           (treesit-parser-set-included-ranges parser (nreverse ranges))
           (push parser reomacs/treesit-rainbow--org-parsers)
           (setq reomacs/treesit-rainbow--parser parser)
           (setq reomacs/treesit-rainbow--depth 1)
           (let ((root (treesit-parser-root-node parser)))
             (when root
               (reomacs/treesit-rainbow--traverse root)))))
       blocks-by-lang)))

  (defun reomacs/treesit-rainbow--highlight ()
    "Parse buffer and apply rainbow overlays."
    ;; Clear existing overlays
    (mapc #'delete-overlay reomacs/treesit-rainbow--overlays)
    (setq reomacs/treesit-rainbow--overlays nil)
    (cond
     ;; Org-mode: highlight embedded source blocks
     ((derived-mode-p 'org-mode)
      (reomacs/treesit-rainbow--highlight-org-blocks))
     ;; Regular treesit buffer
     (t
      (let ((lang (reomacs/treesit-rainbow--get-language)))
        (when lang
          (setq reomacs/treesit-rainbow--parser (treesit-parser-create lang))
          (setq reomacs/treesit-rainbow--depth 1)
          (let ((root (treesit-parser-root-node reomacs/treesit-rainbow--parser)))
            (when root
              (reomacs/treesit-rainbow--traverse root))))))))

Emacs 29+ includes native tree-sitter support via the `treesit` module. This
provides fast syntax highlighting, indentation, and structural navigation via
*-ts-mode variants (python-ts-mode, rust-ts-mode, etc).

Grammars should be provided by your system package manager (e.g., Nix's
treesit-grammars). Emacs auto-discovers grammars in standard locations.

    (defun reomacs/treesit--build-remap-alist ()
      "Build remap alist only for languages with available grammars."
      (->> '((bash sh) c c++ cmake css dockerfile go html java
             (javascript js js2) (json js-json) lua nix python ruby rust toml tsx typescript yaml zig)
           (--mapcat (let* ((modes (if (listp it) it (list it)))
                            (lang (-first-item modes))
                            (ts-mode (intern (format "%s-ts-mode" lang))))
                       (when (reomacs/treesit--grammar-available-p lang)
                         (--map (cons (intern (format "%s-mode" it)) ts-mode) modes))))))

    (setq major-mode-remap-alist (reomacs/treesit--build-remap-alist))

    ;; Rebuild remaps after init in case grammars load lazily

      "Like `vertico-directory-delete-char' but respects TRAMP boundaries.
When in a remote file path, treats `:' `@' `|' and `#' as additional
boundaries beyond `/', so backspace stops at TRAMP components.
If point is right after a delimiter, delete just that delimiter.
If point is after a non-delimiter, delete back to the previous boundary."
      (interactive "p")
      (let* ((file-category (eq 'file (vertico--metadata-get 'category)))
             (no-region (not (and (use-region-p) delete-active-region (= n 1))))
             (in-remote (and file-category
                             no-region
                             (> (point) (minibuffer-prompt-end))
                             (vertico--remote-p
                              (buffer-substring-no-properties
                               (minibuffer-prompt-end) (point))))))
        (cond
         ;; TRAMP path, at a delimiter: delete just the delimiter
         ((and in-remote (memq (char-before) '(?/ ?: ?@ ?| ?#)))
          (with-no-warnings (delete-backward-char n)))
         ;; TRAMP path, at a non-delimiter: delete back to previous boundary
         ((and in-remote (> (point) (minibuffer-prompt-end)))
          (let (found)
            (dotimes (_ n found)
              (save-excursion
                (let ((end (point)))
                  (goto-char (1- end))
                  (if (re-search-backward "[/:@|#]" (minibuffer-prompt-end) t)
                      (progn
                        (delete-region (1+ (point)) end)
                        (setq found t))
                    (delete-region (minibuffer-prompt-end) end)
                    (setq found t)))))
            (unless found
              (with-no-warnings (delete-backward-char n)))))
         ;; Non-remote: default vertico-directory behavior
         (t
          (unless (and no-region (eq (char-before) ?/)
                       (vertico-directory-up n))
            (with-no-warnings (delete-backward-char n))))))))

  ;;;###autoload
  (defun reomacs/dailies-new-day ()
    "Create a new day entry in the dailies file.
Finds the last date entry, creates a new heading for today (or next
workday if already exists), copies previous 'Today' to 'Yesterday'
(or appropriate weekday name), and creates empty 'Today' section."
    (interactive)
    (require 'calendar)
    (-let* ((dailies-buf (find-file-noselect (expand-file-name reomacs/dailies-file)))
            (today (calendar-current-date))
            (today-time (encode-time 0 0 0 (nth 1 today) (nth 0 today) (nth 2 today)))
            (last-date-pos nil)
            (last-date-str nil)
            (last-today-content nil))
      (with-current-buffer dailies-buf
        (goto-char (point-min))
        ;; Find all date headings, keep track of last one
        (while (re-search-forward "^\\*\\* <\\([0-9]\\{4\\}-[0-9]\\{2\\}-[0-9]\\{2\\}\\) \\([A-Za-z]+\\)>" nil t)
          (setq last-date-pos (match-beginning 0))
          (setq last-date-str (match-string 1)))
        (unless (and last-date-pos last-date-str)
          (user-error "No date entries found in dailies file"))
        ;; Go to the last date entry and extract "Today" content
        (goto-char last-date-pos)
        (save-excursion
          (when (re-search-forward "^\\*\\*\\* Today\n" nil t)
            (let ((start (point)))
              (if (re-search-forward "^\\*\\{1,3\\} " nil t)
                  (setq last-today-content
                        (string-trim-right
                         (buffer-substring-no-properties start (match-beginning 0))))
                (setq last-today-content
                      (string-trim-right
                       (buffer-substring-no-properties start (point-max))))))))
        ;; Parse last date and create new entry
        (-let* ((parts (split-string last-date-str "-"))
                (last-time (encode-time 0 0 0
                                        (string-to-number (nth 2 parts))
                                        (string-to-number (nth 1 parts))
                                        (string-to-number (nth 0 parts))))
                (last-is-today-or-later (>= (float-time last-time) (float-time today-time)))
                (new-time (if last-is-today-or-later
                              (reomacs/dailies--next-workday last-time)
                            today-time))
                (new-date-str (format-time-string "<%Y-%m-%d %a>" new-time))
                (yesterday-label (reomacs/dailies--yesterday-label last-time new-time))
                ;; Find where to insert (end of last date subtree, not end of file)
                (insert-pos (save-excursion
                              (goto-char last-date-pos)
                              (org-end-of-subtree t)
                              (point))))
          ;; Go to insert position
          (goto-char insert-pos)
          (skip-chars-backward " \t\n")
          (end-of-line)
          ;; Insert new entry
          (insert "\n\n** " new-date-str "\n\n")
          (insert "*** " yesterday-label "\n")
          (when last-today-content
            (insert last-today-content "\n"))
          (insert "\n*** Today\n\n+ ")
          (switch-to-buffer dailies-buf)
          (message "Created new daily entry for %s" new-date-str)))))

  (defun reomacs/dailies--next-workday (from-date)
    "Return the next workday after FROM-DATE (a time value).
Skips weekends. Could be extended for holidays."
    (-let* ((one-day (* 24 60 60))
            (next-time (time-add from-date one-day))
            (dow (string-to-number (format-time-string "%u" next-time)))) ; 1=Mon, 7=Sun
      ;; Skip weekends
      (while (>= dow 6)
        (setq next-time (time-add next-time one-day))
        (setq dow (string-to-number (format-time-string "%u" next-time))))
      next-time))

  (defun reomacs/dailies--yesterday-label (last-time new-time)
    "Determine the label for the 'yesterday' section.
LAST-TIME and NEW-TIME are Emacs time values.
Returns 'Yesterday', 'Friday', 'Last Week', or the weekday name."
    (-let* ((diff-days (/ (float-time (time-subtract new-time last-time)) 86400))
            (last-dow (format-time-string "%A" last-time))
            (last-dow-num (string-to-number (format-time-string "%u" last-time)))
            (new-dow-num (string-to-number (format-time-string "%u" new-time))))
      (cond
       ((< diff-days 2) "Yesterday")
       ((< diff-days 4) last-dow)     ; Within a few days, use weekday name
       ((= last-dow-num new-dow-num) (format "Last %s" last-dow))  ; Same weekday (e.g., Thu->Tue after holiday)
       ((< diff-days 8) "Last Week")
       (t (format "Last %s" last-dow)))))

  ;; Runner abstraction for sync/async unification
  ;; A runner is a plist with:
  ;;   :fetch (cmd callback) - execute CMD, call CALLBACK with output
  ;;   :emit (line) - output a result line
  ;;   :spawn () - register a pending sub-request
  ;;   :finish () - mark a sub-request complete
  ;;   :date - target date string
  ;;   :username - GitHub username

  (defun reomacs/dailies--run-gh-fetch (runner)
    "Execute GitHub interactions fetch using RUNNER abstraction."
    (-let* (((&plist :date :username :fetch :spawn :finish) runner)
            (next-date (--> (concat date "T00:00:00")
                            (date-to-time it)
                            (time-add it (* 24 60 60))
                            (format-time-string "%Y-%m-%d" it)))
            (date-filter (format "--updated=%s..%s" date next-date)))
      ;; Spawn 4 parallel searches
      (-dotimes 4 (lambda (_) (funcall spawn)))
      ;; Authored PRs
      (funcall fetch
               (format "gh search prs --author=@me %s --json number,repository,url,createdAt,closedAt,state" date-filter)
               (lambda (output)
                 (-some--> output
                   (json-parse-string it :array-type 'list :object-type 'alist)
                   (reomacs/dailies--process-prs runner it))
                 (funcall finish)))
      ;; Reviewed PRs
      (funcall fetch
               (format "gh search prs --reviewed-by=@me %s --json number,repository,url" date-filter)
               (lambda (output)
                 (-some--> output
                   (json-parse-string it :array-type 'list :object-type 'alist)
                   (reomacs/dailies--process-reviews runner it))
                 (funcall finish)))
      ;; Authored issues
      (funcall fetch
               (format "gh search issues --author=@me %s --json number,repository,url,createdAt" date-filter)
               (lambda (output)
                 (-some--> output
                   (json-parse-string it :array-type 'list :object-type 'alist)
                   (reomacs/dailies--process-issues runner it))
                 (funcall finish)))
      ;; Branch pushes
      (funcall fetch
               (list "gh" "api" (format "/users/%s/events?per_page=100" username)
                     "--jq" (format "[.[] | select(.type == \"PushEvent\" and (.created_at | startswith(\"%s\")))] | unique_by(.repo.name + .payload.ref) | .[] | {repo: .repo.name, branch: .payload.ref}"
                                    date))
               (lambda (output)
                 (-some--> (string-trim output)
                   (unless (string-empty-p it) it)
                   (split-string it "\n" t)
                   (reomacs/dailies--process-pushes runner it))
                 (funcall finish)))))

  (defun reomacs/dailies--process-prs (runner prs)
    "Process authored PRs using RUNNER."
    (-let [(&plist :date :username :fetch :emit :spawn :finish) runner]
      (--each prs
        (-let* [((&alist 'url 'number 'createdAt 'closedAt
                         'repository (&alist 'name 'nameWithOwner)) it)
                (display-name (reomacs/dailies--repo-display-name nameWithOwner name))
                (props (list :url url :display-name display-name :number number))]
          (cond
           ((reomacs/dailies--date-on-day-p createdAt date)
            (funcall emit (reomacs/dailies--format-event 'pr-create props)))
           ((reomacs/dailies--date-on-day-p closedAt date)
            (funcall spawn)
            (funcall fetch
                     (list "gh" "api" (format "/repos/%s/issues/%s/events" nameWithOwner number)
                           "--jq" "[.[] | select(.event == \"closed\" or .event == \"merged\")] | last | {actor: .actor.login, event: .event}")
                     (lambda (output)
                       (-let* [((&alist 'actor 'event) (ignore-errors (json-parse-string (string-trim output) :object-type 'alist)))
                               (event-sym (if (string= event "merged") 'pr-merge 'pr-close))
                               (props-final (-if-let (other (and (not (string= actor username)) actor))
                                                (plist-put (copy-sequence props) :actor other)
                                              props))]
                         (funcall emit (reomacs/dailies--format-event event-sym props-final)))
                       (funcall finish)))))))))

  (defun reomacs/dailies--process-reviews (runner prs)
    "Process reviewed PRs using RUNNER."
    (-let [(&plist :date :username :fetch :emit :spawn :finish) runner]
      (--each prs
        (-let* [((&alist 'url 'number 'repository (&alist 'name 'nameWithOwner)) it)
                (display-name (reomacs/dailies--repo-display-name nameWithOwner name))]
          (funcall spawn)
          (funcall fetch
                   (list "gh" "api" (format "/repos/%s/pulls/%s/reviews" nameWithOwner number)

  (defun reomacs/dailies--gh-async (command callback)
    "Run gh COMMAND asynchronously, call CALLBACK with output string.
COMMAND can be a string (split by shell rules) or a list of args.
Sets TERM=dumb and sends EOF to prevent gh from blocking on terminal."
    (-let* ((process-environment (cons "TERM=dumb" process-environment))
            (buf (generate-new-buffer " *gh-async*"))
            (cmd-list (if (listp command) command
                        (split-string-shell-command command)))
            (proc (make-process
                   :name "gh-async"
                   :buffer buf
                   :connection-type 'pipe
                   :command cmd-list
                   :sentinel
                   (lambda (proc event)
                     (cond
                      ((string-match-p "finished" event)
                       (unwind-protect
                           (funcall callback (with-current-buffer (process-buffer proc)
                                               (buffer-string)))
                         (kill-buffer (process-buffer proc))))
                      ((string-match-p "\\(exited\\|failed\\)" event)
                       (ignore-errors (kill-buffer (process-buffer proc)))))))))
      (process-send-eof proc)
      proc))

    "Create an async runner for DATE with MARKER for insertion.
Returns a plist with runner functions using async state management."
    (-let* ((state (make-hash-table :test 'eq)))
      (puthash :pending 0 state)
      (puthash :marker marker state)
      (list :date date
            :username nil  ; filled in by initial fetch
            :state state
            :fetch (lambda (cmd callback)
                     (reomacs/dailies--gh-async cmd callback))
            :emit (lambda (line)
                    (-when-let (m (gethash :marker state))
                      (when (marker-buffer m)
                        (with-current-buffer (marker-buffer m)
                          (save-excursion
                            (goto-char m)
                            (insert-before-markers line "\n"))))))
            :spawn (lambda ()
                     (puthash :pending (1+ (gethash :pending state)) state))
            :finish (lambda ()
                      (puthash :pending (1- (gethash :pending state)) state)
                      (when (<= (gethash :pending state) 0)
                        (-when-let (m (gethash :marker state))
                          (when (marker-buffer m)
                            (with-current-buffer (marker-buffer m)
                              (save-excursion
                                (goto-char m)
                                (when (looking-at (regexp-quote (concat "+ " reomacs/dailies--gh-placeholder)))
                                  (delete-region (point) (line-beginning-position 2))))))

(defun reomacs/wdired-create-new-files-advice (orig-fun &rest args)
  "Advise `wdired-finish-edit` to create new files and refresh.

This function is designed to be used as `:around` advice. It scans the
`wdired` buffer for lines that do not correspond to existing file
entries (i.e., they lack the `dired-filename` text property).

For each such line found:
1. It parses the line as a new file path.
2. It creates any non-existent parent directories recursively.
3. It creates the new file (as an empty file).
4. It removes the line from the buffer.

After creating all new files, it calls the original `wdired-finish-edit`
function. If any files were created by this advice, it then calls
`revert-buffer` to refresh the dired listing."
  (let ((files-created-p nil))
    (save-excursion
      ;; Go to the beginning of the buffer to start our scan.
      (goto-char (point-min))

      ;; 1. First, we collect all the lines that represent new files.
      ;; We do this first to avoid issues with modifying the buffer while iterating over it.
      (let ((new-entries '()))
        (while (not (eobp))
          (let ((line-start (line-beginning-position))
                (line-end (line-end-position)))
            ;; A line for a new file won't have the 'dired-filename' property.
            (unless (get-text-property line-start 'dired-filename)
              (let ((filename (string-trim (buffer-substring-no-properties line-start line-end))))
                ;; We only care about lines that actually contain non-whitespace text.
                (when (string-match-p "\\S-" filename)
                  ;; We push the filename and its start/end positions onto the list.
                  (push (list filename line-start line-end) new-entries))))
            (forward-line 1)))

        ;; 2. Now, process the collected new files.
        ;; We iterate through the list we just built. The list is naturally in
        ;; reverse order of appearance, which is perfect for deleting lines
        ;; from the buffer without invalidating the position markers of subsequent items.
        (when new-entries
          (setq files-created-p t)) ; Mark that we are creating files.
        (dolist (entry new-entries)
          (let* ((filename (nth 0 entry))
                 (start-pos (nth 1 entry))
                 (end-pos (nth 2 entry))
                 ;; Construct the full path relative to the dired buffer's directory.
                 (full-path (expand-file-name filename default-directory))
                 (directory (file-name-directory full-path)))

            ;; Create parent directories if they don't already exist.
            ;; The 't' argument to `make-directory` means "create parents as needed".
            (when (and directory (not (file-directory-p directory)))
              (make-directory directory t)
              (message "Created directory: %s" (file-name-sans-trailing-slash directory)))

            ;; Create the file itself if it doesn't exist.
            ;; Using `with-temp-buffer` and `write-file` is a clean way to "touch" a file.
            (unless (file-exists-p full-path)
              (with-temp-buffer
                (write-file full-path))
              (message "Created file: %s" full-path))

            ;; Finally, remove the processed line from the wdired buffer so the original
            ;; function doesn't see it and get confused.
            (goto-char start-pos)
            ;; We delete the region including the newline character.
            (delete-region start-pos (1+ end-pos)))))))

    ;; 3. With our work done, call the original function to let it handle its business.
    (apply orig-fun args)

    ;; 4. If we created any files, refresh the dired buffer to show them.
    (when files-created-p
      (revert-buffer)
      (message "Dired buffer refreshed to show new files.")))

;; Add our function as "around" advice to `wdired-finish-edit`.
;; This means our function wraps the original one.
(advice-add 'wdired-finish-edit :around #'reomacs/wdired-create-new-files-advice)

  (defvar-local reomodeline--jj-info nil)
  (defvar-local reomodeline--jj-root nil)

  (defvar-local reomodeline--jj-timer nil)

  (defun reomodeline--jj-root ()
    "Get cached jj root, or detect it (skipping remote buffers)."
    (or reomodeline--jj-root
        (setq reomodeline--jj-root
              (and (not (file-remote-p default-directory))
                   (locate-dominating-file default-directory ".jj")))))

  (defun reomodeline--jj-start-process ()
    "Start async jj process to get current change id."
    (when-let ((root (reomodeline--jj-root)))
      ;; Cancel any existing process
      (when (process-live-p reomodeline--jj-proc)
        (delete-process reomodeline--jj-proc))
      (let* ((default-directory root)
             (buf (current-buffer))
             (proc-buf (generate-new-buffer " *reomodeline-jj*")))
        (setq reomodeline--jj-proc
              (make-process
               :name "reomodeline-jj" :buffer proc-buf :noquery t
               :command '("jj" "log" "-r" "@" "--no-graph" "-T" "change_id.short(8)")
               :sentinel
               (lambda (proc _event)
                 (when (memq (process-status proc) '(exit signal))
                   (unwind-protect
                       (when (and (= (process-exit-status proc) 0)
                                  (buffer-live-p buf))
                         (let ((output (with-current-buffer (process-buffer proc)
                                         (string-trim (buffer-string)))))
                           (with-current-buffer buf
                             (setq reomodeline--jj-info
                                   (unless (string-empty-p output) output))
                             (force-mode-line-update))))
                     (when (buffer-live-p (process-buffer proc))
                       (kill-buffer (process-buffer proc)))))))))))

    "Debounced async jj update."
    (when reomodeline--jj-timer

          (run-with-idle-timer 0.2 nil #'reomodeline--jj-start-process)))

  (defun reomodeline--jj-cleanup ()

    (when reomodeline--jj-timer
      (cancel-timer reomodeline--jj-timer)))

  (add-hook 'find-file-hook #'reomodeline-update-jj)

    "Memoize a value if the key is a remote path."
    (if (and key (file-remote-p key))
        (if-let ((current (assoc key (symbol-value cache))))
            (cdr current)
          (let ((current (apply orig-fn args)))
            (set cache (cons (cons key current) (symbol-value cache)))
            current))
      (apply orig-fn args)))

  ;; Memoize project-current
  (defvar project-current-cache nil)
  (defun memoize-project-current (orig &optional prompt directory)
    (memoize-remote (or directory project-current-directory-override default-directory)
                    'project-current-cache orig prompt directory))
  (advice-add 'project-current :around #'memoize-project-current)

  (use-package msgpack)

  ;; Fallback: proxy sudo through SSH for hosts without passwordless sudo.
  ;; Use `/sudo:root@HOST:/path` — TRAMP routes sudo through an SSH shell.
  ;; (add-to-list 'tramp-default-proxies-alist
  ;;              '("jeeves\\.lan" "root" "/ssh:jeeves@jeeves.lan:"))

  ;; Sudo composition — `/rpc:root@HOST`: `SSH`es as the configured user
  ;; and runs `tramp-rpc-server` under `sudo`, giving root-level RPC access.
  ;; Requires passwordless `sudo` for `tramp-rpc-server` on the remote host.
  (defvar reomacs/tramp-rpc-sudo-compose-hosts '(("jeeves\\.lan" . "jeeves"))
    "Alist of (HOST-REGEXP . SSH-USER) for rpc sudo composition.
When connecting as root via rpc to a matching host, SSH connects
as SSH-USER and runs tramp-rpc-server under sudo.")

  (defun reomacs/tramp-rpc--sudo-match (host)
    "Return the SSH user for HOST if it matches a sudo-compose entry."
    (--some (when (string-match-p (car it) host)
              (cdr it))
            reomacs/tramp-rpc-sudo-compose-hosts))

  (defun reomacs/tramp-rpc-sudo-connect-advice (orig-fn vec)
    "Advise `tramp-rpc--connect' for sudo composition.
When user is root and host matches `reomacs/tramp-rpc-sudo-compose-hosts',
SSH as the real user and launch tramp-rpc-server under sudo.
The binary path's ~/ is expanded as the SSH user, not root."
    (let ((host (tramp-file-name-host vec)))
      (-if-let (real-user (and (equal (tramp-file-name-user vec) "root")
                               (reomacs/tramp-rpc--sudo-match host)))
          ;; Build an SSH vec with the real user for ControlMaster/socket paths
          (let ((ssh-vec (copy-tramp-file-name vec)))
            (setf (tramp-file-name-user ssh-vec) real-user)
            ;; 1. ControlMaster as the real SSH user
            (tramp-rpc--ensure-controlmaster-directory)
            (when tramp-rpc-use-controlmaster
              (tramp-rpc--establish-controlmaster ssh-vec))
            ;; 2. Start server: SSH as real-user, sudo the binary,
            ;;    but use original vec (user=root) for TRAMP naming/storage
            (let* ((binary-path (or tramp-rpc-deploy-remote-binary-path
                                    "tramp-rpc-server"))
                   ;; Expand ~/ as real user's home so sudo doesn't
                   ;; resolve it as root's home
                   (expanded-binary
                    (replace-regexp-in-string
                     "\\`~/" (format "~%s/" real-user) binary-path))
                   (port (tramp-file-name-port vec))
                   (proxyjump (tramp-rpc--hops-to-proxyjump vec))
                   (ssh-args
                    (append
                     (list "ssh")
                     tramp-rpc-ssh-args
                     (list "-l" real-user)
                     (when port (list "-p" (number-to-string port)))
                     (when proxyjump (list "-J" proxyjump))
                     (list "-o" "BatchMode=yes")
                     (list "-o" "StrictHostKeyChecking=accept-new")
                     (--mapcat (list "-o" it) tramp-rpc-ssh-options)
                     (when tramp-rpc-use-controlmaster
                       (list "-o" "ControlMaster=auto"
                             "-o" (format "ControlPath=%s"
                                          (tramp-rpc--controlmaster-socket-path ssh-vec))
                             "-o" (format "ControlPersist=%s"
                                          tramp-rpc-controlmaster-persist)))
                     (list host (format "sudo %s" expanded-binary))))
                   (process-name (tramp-get-connection-name vec))
                   (buffer-name (tramp-buffer-name vec))
                   (buffer (get-buffer-create buffer-name))
                   process)
              (with-current-buffer buffer
                (erase-buffer)
                (set-buffer-multibyte nil))
              (let ((process-connection-type nil))
                (setq process (apply #'start-process process-name buffer ssh-args)))
              (set-process-query-on-exit-flag process nil)
              (set-process-coding-system process 'binary 'binary)
              (set-process-filter process #'tramp-rpc--connection-filter)
              (tramp-rpc--set-connection vec process buffer)
              (process-put process :tramp-rpc-vec vec)
              (condition-case err
                  (let ((response (tramp-rpc--call vec "system.info" nil)))
                    (unless response
                      (signal 'remote-file-error
                              (list "sudo RPC server not responding on" host))))
                (remote-file-error
                 (tramp-rpc--remove-connection vec)
                 (signal (car err) (cdr err))))
              (tramp-set-connection-property process "connected" t)
              (tramp-rpc--get-connection vec)))
        (funcall orig-fn vec))))
