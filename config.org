  ;;; -*- lexical-binding: t -*-

  ;; Strategy: Always load pre-tangled config.el directly.
  ;; This avoids loading built-in org before straight can install a custom version.
  ;; Tangling happens automatically on save (see auto-tangle in config.org).
  ;; If config.el is missing, we fall back to tangling with built-in org.

  (let ((config-org (expand-file-name "config.org" user-emacs-directory))
        (config-el (expand-file-name "config.el" user-emacs-directory)))
    (cond
     ;; Prefer pre-tangled config.el (allows custom org version)
     ((file-exists-p config-el)
      (load-file config-el))
     ;; Fallback: tangle with built-in org (first-time bootstrap only)
     ((file-exists-p config-org)
      (require 'org)
      (let ((org-confirm-babel-evaluate nil))
        (org-babel-tangle-file config-org))
      (load-file config-el))
     (t
      (warn "Neither config.el nor config.org found in %s" user-emacs-directory))))
#+end_src

  (defun reomacs/tangle-config-org ()
    "Tangle config.org automatically if in `user-emacs-directory`"
    (when (string-equal (buffer-file-name)
                        (expand-file-name "config.org" user-emacs-directory))
      (let ((org-confirm-babel-evaluate nil))
        (org-babel-tangle))))

  (add-hook 'after-save-hook #'reomacs/tangle-config-org)
  (comment
   (reomacs/tangle-config-org))

  (defvar reomacs/post-init-hooks nil
    "Functions to run after init and after config reload.")

  (defun reomacs/run-post-init-hooks ()
    "Run all functions in `reomacs/post-init-hooks'."
    (run-hooks 'reomacs/post-init-hooks))

  (add-hook 'emacs-startup-hook #'reomacs/run-post-init-hooks)

  (defvar reomacs/config-loaded-p nil
    "Non-nil after initial config load. Used to skip re-initialization.")

  (defun reomacs/reload-config ()
    "Tangle config.org and reload config.el.
Skips display-touching initialization on reload to avoid PGTK crashes."
    (interactive)
    (let ((org-confirm-babel-evaluate nil)
          (reomacs/config-loaded-p t))  ; Signal we're reloading
      (org-babel-tangle-file (expand-file-name "config.org" user-emacs-directory))
      (load-file (expand-file-name "config.el" user-emacs-directory))
      (reomacs/run-post-init-hooks)
      (message "Config tangled and reloaded.")))

  (add-hook 'emacs-startup-hook (lambda () (setq reomacs/config-loaded-p t)) 100)
#+end_src

#+begin_src emacs-lisp
  ;; Toggle for global auto-completion visibility
  (defvar reomacs/corfu-auto-enabled t
    "Whether Corfu auto-completion popup is enabled globally.")

  (defun reomacs/corfu-toggle-auto ()
    "Toggle Corfu auto-completion on/off globally."
    (interactive)
    (setq reomacs/corfu-auto-enabled (not reomacs/corfu-auto-enabled))
    (setq corfu-auto reomacs/corfu-auto-enabled)
    (message "Corfu auto-completion %s" (if reomacs/corfu-auto-enabled "enabled" "disabled")))

  (use-package corfu
    :custom
    ;; Auto-completion (toggleable via reomacs/corfu-toggle-auto)
    (corfu-auto t)
    (corfu-auto-delay 0)
    (corfu-auto-prefix 1)

    ;; Neovim-like behavior: NO preview insertion, just visual ghost text
    (corfu-preview-current nil)      ; Disable preview (prevents any insertion)
    (corfu-preselect 'prompt)        ; Don't preselect any candidate initially
    (corfu-on-exact-match nil)       ; Don't auto-insert on exact match

    ;; Cycling through candidates
    (corfu-cycle t)

    ;; Popup behavior
    (corfu-count 10)
    (corfu-scroll-margin 2)
    (corfu-quit-no-match 'separator)

    :init
    (global-corfu-mode)

    :config
    ;; C-n: Move to next candidate (or trigger popup if hidden)
    (defun reomacs/corfu-next-or-complete ()
      "Move to next Corfu candidate, triggering completion if not active."
      (interactive)
      (if completion-in-region-mode
          (corfu-next 1)
        ;; Not in completion - trigger it
        (when (thing-at-point 'symbol)
          (completion-at-point))))

    ;; C-p: Move to previous candidate (or trigger popup if hidden)
    (defun reomacs/corfu-previous-or-complete ()
      "Move to previous Corfu candidate, triggering completion if not active."
      (interactive)
      (if completion-in-region-mode
          (corfu-previous 1)
        ;; Not in completion - trigger it and go to last
        (when (thing-at-point 'symbol)
          (completion-at-point)
          (when completion-in-region-mode
            (corfu-last)))))

    ;; Keybindings in corfu-map (when popup is active)
    (define-key corfu-map (kbd "C-n") #'corfu-next)
    (define-key corfu-map (kbd "C-p") #'corfu-previous)

  ;;;###autoload
  (defun reomacs/dailies-new-day ()
    "Create a new day entry in the dailies file.
Finds the last date entry, creates a new heading for today (or next
workday if already exists), copies previous 'Today' to 'Yesterday'
(or appropriate weekday name), and creates empty 'Today' section."
    (interactive)
    (require 'calendar)
    (-let* ((dailies-buf (find-file-noselect (expand-file-name reomacs/dailies-file)))
            (today (calendar-current-date))
            (today-time (encode-time 0 0 0 (nth 1 today) (nth 0 today) (nth 2 today)))
            (last-date-pos nil)
            (last-date-str nil)
            (last-today-content nil))
      (with-current-buffer dailies-buf
        (goto-char (point-min))
        ;; Find all date headings, keep track of last one
        (while (re-search-forward "^\\*\\* <\\([0-9]\\{4\\}-[0-9]\\{2\\}-[0-9]\\{2\\}\\) \\([A-Za-z]+\\)>" nil t)
          (setq last-date-pos (match-beginning 0))
          (setq last-date-str (match-string 1)))
        (unless (and last-date-pos last-date-str)
          (user-error "No date entries found in dailies file"))
        ;; Go to the last date entry and extract "Today" content
        (goto-char last-date-pos)
        (save-excursion
          (when (re-search-forward "^\\*\\*\\* Today\n" nil t)
            (let ((start (point)))
              (if (re-search-forward "^\\*\\{1,3\\} " nil t)
                  (setq last-today-content
                        (string-trim-right
                         (buffer-substring-no-properties start (match-beginning 0))))
                (setq last-today-content
                      (string-trim-right
                       (buffer-substring-no-properties start (point-max))))))))
        ;; Parse last date and create new entry
        (-let* ((parts (split-string last-date-str "-"))
                (last-time (encode-time 0 0 0
                                        (string-to-number (nth 2 parts))
                                        (string-to-number (nth 1 parts))
                                        (string-to-number (nth 0 parts))))
                (last-is-today-or-later (>= (float-time last-time) (float-time today-time)))
                (new-time (if last-is-today-or-later
                              (reomacs/dailies--next-workday last-time)
                            today-time))
                (new-date-str (format-time-string "<%Y-%m-%d %a>" new-time))
                (yesterday-label (reomacs/dailies--yesterday-label last-time new-time))
                ;; Find where to insert (end of last date subtree, not end of file)
                (insert-pos (save-excursion
                              (goto-char last-date-pos)
                              (org-end-of-subtree t)
                              (point))))
          ;; Go to insert position
          (goto-char insert-pos)
          (skip-chars-backward " \t\n")
          (end-of-line)
          ;; Insert new entry
          (insert "\n\n** " new-date-str "\n\n")
          (insert "*** " yesterday-label "\n")
          (when last-today-content
            (insert last-today-content "\n"))
          (insert "\n*** Today\n\n+ ")
          (switch-to-buffer dailies-buf)
          (message "Created new daily entry for %s" new-date-str)))))

  (defun reomacs/dailies--next-workday (from-date)
    "Return the next workday after FROM-DATE (a time value).
Skips weekends. Could be extended for holidays."
    (-let* ((one-day (* 24 60 60))
            (next-time (time-add from-date one-day))
            (dow (string-to-number (format-time-string "%u" next-time)))) ; 1=Mon, 7=Sun
      ;; Skip weekends
      (while (>= dow 6)
        (setq next-time (time-add next-time one-day))
        (setq dow (string-to-number (format-time-string "%u" next-time))))
      next-time))

  (defun reomacs/dailies--yesterday-label (last-time new-time)
    "Determine the label for the 'yesterday' section.
LAST-TIME and NEW-TIME are Emacs time values.
Returns 'Yesterday', 'Friday', 'Last Week', or the weekday name."
    (-let* ((diff-days (/ (float-time (time-subtract new-time last-time)) 86400))
            (last-dow (format-time-string "%A" last-time))
            (last-dow-num (string-to-number (format-time-string "%u" last-time)))
            (new-dow-num (string-to-number (format-time-string "%u" new-time))))
      (cond
       ((< diff-days 2) "Yesterday")
       ((< diff-days 4) last-dow)     ; Within a few days, use weekday name
       ((= last-dow-num new-dow-num) (format "Last %s" last-dow))  ; Same weekday (e.g., Thu->Tue after holiday)
       ((< diff-days 8) "Last Week")
       (t (format "Last %s" last-dow)))))

  ;;   :date - target date string
  ;;   :username - GitHub username

  (defun reomacs/dailies--run-gh-fetch (runner)
    "Execute GitHub interactions fetch using RUNNER abstraction."
    (-let* (((&plist :date :username :fetch :spawn :finish) runner)
            (next-date (--> (concat date "T00:00:00")
                            (date-to-time it)
                            (time-add it (* 24 60 60))
                            (format-time-string "%Y-%m-%d" it)))
            (date-filter (format "--updated=%s..%s" date next-date)))
      ;; Spawn 4 parallel searches
      (-dotimes 4 (lambda (_) (funcall spawn)))
      ;; Authored PRs
      (funcall fetch
               (format "gh search prs --author=@me %s --json number,repository,url,createdAt,closedAt,state" date-filter)
               (lambda (output)
                 (-some--> output
                   (json-parse-string it :array-type 'list :object-type 'alist)
                   (reomacs/dailies--process-prs runner it))
                 (funcall finish)))
      ;; Reviewed PRs
      (funcall fetch
               (format "gh search prs --reviewed-by=@me %s --json number,repository,url" date-filter)
               (lambda (output)
                 (-some--> output
                   (json-parse-string it :array-type 'list :object-type 'alist)
                   (reomacs/dailies--process-reviews runner it))
                 (funcall finish)))
      ;; Authored issues
      (funcall fetch
               (format "gh search issues --author=@me %s --json number,repository,url,createdAt" date-filter)
               (lambda (output)
                 (-some--> output
                   (json-parse-string it :array-type 'list :object-type 'alist)
                   (reomacs/dailies--process-issues runner it))
                 (funcall finish)))
      ;; Branch pushes
      (funcall fetch
               (list "gh" "api" (format "/users/%s/events?per_page=100" username)
                     "--jq" (format "[.[] | select(.type == \"PushEvent\" and (.created_at | startswith(\"%s\")))] | unique_by(.repo.name + .payload.ref) | .[] | {repo: .repo.name, branch: .payload.ref}"
                                    date))
               (lambda (output)
                 (-some--> (string-trim output)
                   (unless (string-empty-p it) it)
                   (split-string it "\n" t)
                   (reomacs/dailies--process-pushes runner it))
                 (funcall finish)))))

  (defun reomacs/dailies--process-prs (runner prs)
    "Process authored PRs using RUNNER."
    (-let [(&plist :date :username :fetch :emit :spawn :finish) runner]
      (--each prs
        (-let* [((&alist 'url 'number 'createdAt 'closedAt
                         'repository (&alist 'name 'nameWithOwner)) it)
                (display-name (reomacs/dailies--repo-display-name nameWithOwner name))
                (props (list :url url :display-name display-name :number number))]
          (cond
           ((reomacs/dailies--date-on-day-p createdAt date)
            (funcall emit (reomacs/dailies--format-event 'pr-create props)))
           ((reomacs/dailies--date-on-day-p closedAt date)
            (funcall spawn)
            (funcall fetch
                     (list "gh" "api" (format "/repos/%s/issues/%s/events" nameWithOwner number)
                           "--jq" "[.[] | select(.event == \"closed\" or .event == \"merged\")] | last | {actor: .actor.login, event: .event}")
                     (lambda (output)
                       (-let* [((&alist 'actor 'event) (ignore-errors (json-parse-string (string-trim output) :object-type 'alist)))
                               (event-sym (if (string= event "merged") 'pr-merge 'pr-close))
                               (props-final (-if-let (other (and (not (string= actor username)) actor))
                                                (plist-put (copy-sequence props) :actor other)
                                              props))]
                         (funcall emit (reomacs/dailies--format-event event-sym props-final)))
                       (funcall finish)))))))))

    "Process reviewed PRs using RUNNER."
    (-let [(&plist :date :username :fetch :emit :spawn :finish) runner]

        (-let* [((&alist 'url 'number 'repository (&alist 'name 'nameWithOwner)) it)
                (display-name (reomacs/dailies--repo-display-name nameWithOwner name))]

                                        username date))

                     (-let [count (-> output string-trim string-to-number)]
                       (when (> count 0)

  (defun reomacs/dailies--gh-async (command callback)
    "Run gh COMMAND asynchronously, call CALLBACK with output string.
COMMAND can be a string (split by shell rules) or a list of args.
Sets TERM=dumb and sends EOF to prevent gh from blocking on terminal."
    (-let* ((process-environment (cons "TERM=dumb" process-environment))
            (buf (generate-new-buffer " *gh-async*"))
            (cmd-list (if (listp command) command
                        (split-string-shell-command command)))
            (proc (make-process
                   :name "gh-async"
                   :buffer buf
                   :connection-type 'pipe
                   :command cmd-list
                   :sentinel
                   (lambda (proc event)
                     (cond
                      ((string-match-p "finished" event)
                       (unwind-protect
                           (funcall callback (with-current-buffer (process-buffer proc)
                                               (buffer-string)))
                         (kill-buffer (process-buffer proc))))
                      ((string-match-p "\\(exited\\|failed\\)" event)
                       (ignore-errors (kill-buffer (process-buffer proc)))))))))
      (process-send-eof proc)
      proc))

    "Create an async runner for DATE with MARKER for insertion.
Returns a plist with runner functions using async state management."
    (-let* ((state (make-hash-table :test 'eq)))
      (puthash :pending 0 state)
      (puthash :marker marker state)
      (list :date date
            :username nil  ; filled in by initial fetch
            :state state
            :fetch (lambda (cmd callback)
                     (reomacs/dailies--gh-async cmd callback))
            :emit (lambda (line)
                    (-when-let (m (gethash :marker state))
                      (when (marker-buffer m)
                        (with-current-buffer (marker-buffer m)
                          (save-excursion
                            (goto-char m)
                            (insert-before-markers line "\n"))))))
            :spawn (lambda ()
                     (puthash :pending (1+ (gethash :pending state)) state))
            :finish (lambda ()
                      (puthash :pending (1- (gethash :pending state)) state)
                      (when (<= (gethash :pending state) 0)
                        (-when-let (m (gethash :marker state))
                          (when (marker-buffer m)
                            (with-current-buffer (marker-buffer m)
                              (save-excursion
                                (goto-char m)

(defun reomacs/wdired-create-new-files-advice (orig-fun &rest args)
  "Advise `wdired-finish-edit` to create new files and refresh.

This function is designed to be used as `:around` advice. It scans the
`wdired` buffer for lines that do not correspond to existing file
entries (i.e., they lack the `dired-filename` text property).

For each such line found:
1. It parses the line as a new file path.
2. It creates any non-existent parent directories recursively.
3. It creates the new file (as an empty file).
4. It removes the line from the buffer.

After creating all new files, it calls the original `wdired-finish-edit`
function. If any files were created by this advice, it then calls
`revert-buffer` to refresh the dired listing."
  (let ((files-created-p nil))
    (save-excursion
      ;; Go to the beginning of the buffer to start our scan.
      (goto-char (point-min))

      ;; 1. First, we collect all the lines that represent new files.
      ;; We do this first to avoid issues with modifying the buffer while iterating over it.
      (let ((new-entries '()))
        (while (not (eobp))
          (let ((line-start (point-at-bol))
                (line-end (point-at-eol)))
            ;; A line for a new file won't have the 'dired-filename' property.
            (unless (get-text-property line-start 'dired-filename)
              (let ((filename (string-trim (buffer-substring-no-properties line-start line-end))))
                ;; We only care about lines that actually contain non-whitespace text.
                (when (string-match-p "\\S-" filename)
                  ;; We push the filename and its start/end positions onto the list.
                  (push (list filename line-start line-end) new-entries))))
            (forward-line 1)))

        ;; 2. Now, process the collected new files.
        ;; We iterate through the list we just built. The list is naturally in
        ;; reverse order of appearance, which is perfect for deleting lines
        ;; from the buffer without invalidating the position markers of subsequent items.
        (when new-entries
          (setq files-created-p t)) ; Mark that we are creating files.
        (dolist (entry new-entries)
          (let* ((filename (nth 0 entry))
                 (start-pos (nth 1 entry))
                 (end-pos (nth 2 entry))
                 ;; Construct the full path relative to the dired buffer's directory.
                 (full-path (expand-file-name filename default-directory))
                 (directory (file-name-directory full-path)))

            ;; Create parent directories if they don't already exist.
            ;; The 't' argument to `make-directory` means "create parents as needed".
            (when (and directory (not (file-directory-p directory)))
              (make-directory directory t)
              (message "Created directory: %s" (file-name-sans-trailing-slash directory)))

            ;; Create the file itself if it doesn't exist.
            ;; Using `with-temp-buffer` and `write-file` is a clean way to "touch" a file.
            (unless (file-exists-p full-path)
              (with-temp-buffer
                (write-file full-path))
              (message "Created file: %s" full-path))

            ;; Finally, remove the processed line from the wdired buffer so the original
            ;; function doesn't see it and get confused.
            (goto-char start-pos)
            ;; We delete the region including the newline character.
            (delete-region start-pos (1+ end-pos)))))))

    ;; 3. With our work done, call the original function to let it handle its business.
    (apply orig-fun args)

    ;; 4. If we created any files, refresh the dired buffer to show them.
    (when files-created-p
      (revert-buffer)
      (message "Dired buffer refreshed to show new files.")))

;; Add our function as "around" advice to `wdired-finish-edit`.
;; This means our function wraps the original one.
(advice-add 'wdired-finish-edit :around #'reomacs/wdired-create-new-files-advice)

;; To remove this advice later, you can run:
;; (advice-remove 'wdired-finish-edit #'my-wdired-create-new-files-advice)
#+end_src

  (defvar reomodeline-evil-state-colors
    '((normal . "#ff5555") (insert . "#50fa7b") (visual . "#8be9fd")

      (emacs . "#bd93f9")))

  ;; Detailed state/substate -> name mapping (inspired by neovim heirline)
  ;; Format: (state . name) or ((state . substate) . name)
  (defvar reomodeline-evil-state-names
    '(;; Normal state variants
      (normal . "N")
      ((normal . nil) . "N")
      ;; Insert state variants
      (insert . "I")
      ((insert . completion) . "Ic")    ; completion active
      ;; Visual state variants (evil-visual-selection: char, line, block, screen-line)
      (visual . "V")
      ((visual . char) . "V")
      ((visual . line) . "V_")
      ((visual . block) . "^V")
      ((visual . screen-line) . "Vs")
      ;; Replace state variants
      (replace . "R")
      ((replace . nil) . "R")
      ;; Operator-pending state
      (operator . "O")
      ((operator . nil) . "O")
      ((operator . char) . "Oc")
      ((operator . line) . "Ol")
      ((operator . block) . "Ob")
      ;; Motion state
      (motion . "M")
      ;; Emacs state
      (emacs . "E")))

  (defun reomodeline--get-visual-subtype ()
    "Get visual selection type: char, line, block, or screen-line."
    (when (and (eq evil-state 'visual)
               (bound-and-true-p evil-visual-selection))
      evil-visual-selection))

  (defun reomodeline--get-operator-subtype ()
    "Get operator-pending motion type if available."
    (when (eq evil-state 'operator)
      (bound-and-true-p evil-this-type)))

  (defun reomodeline--in-completion-p ()
    "Check if completion is active in insert mode."
    (and (eq evil-state 'insert)
         (or (bound-and-true-p company-candidates)
             (bound-and-true-p corfu--candidates)
             (minibufferp))))

  (defun reomodeline-vimode ()
    (when (bound-and-true-p evil-mode)
      (let* ((state evil-state)
             (subtype (cond
                       ((eq state 'visual) (reomodeline--get-visual-subtype))
                       ((eq state 'operator) (reomodeline--get-operator-subtype))
                       ((eq state 'insert) (when (reomodeline--in-completion-p) 'completion))
                       (t nil)))
             (key (if subtype (cons state subtype) state))
             (name (or (alist-get key reomodeline-evil-state-names nil nil #'equal)
                       (alist-get state reomodeline-evil-state-names "?")))

                           (_ name))))
